\section{Major Research and Engineering Work Packages}

\subsection{1. Mathematical Modeling and Optimization Formulation}
\begin{itemize}
  \item \textbf{Formulation of Optimization Objectives:}
  \begin{itemize}
    \item \textbf{Coverage Maximization $C(S)$:} Total package-version and repository reach achieved by protecting maintainer set $S$.
    \item \textbf{Additive Scenario Exposure $H(S)$:} Transitive risk mass and downstream blast-radius mitigation.
    \item \textbf{All-Source Exclusion $P(S)$:} Defense-in-depth metric ensuring all maintainers authorized to publish a package are protected.
  \end{itemize}
  \item \textbf{Quorum and Imperfect Mitigation Extensions (Task 13 \& 14):}
  \begin{itemize}
    \item Formulated generalized $k$-of-$n$ quorum exclusion $P_k(S)$, modeling multi-signature and consensus maintainer release requirements.
    \item Formulated imperfect intervention efficacy $H_\eta(S)$, quantifying realistic partial adherence to security controls (MFA, hardware keys).
    \item Mathematical proof and numerical implementation of strict zero-budget boundary conditions.
  \end{itemize}
  \item \textbf{Pareto Optimization and Scalarization (Task 07):}
  \begin{itemize}
    \item Multi-objective scalarization $\alpha C(S) + (1-\alpha) H(S)$ evaluated across fine-grained grid $\alpha \in [0.0, 1.0]$.
    \item Discovery of the $\alpha = 0.50$ sweet spot capturing 70\% of maximum coverage while retaining 31.2\% of exposure mitigation.
    \item Heterogeneous risk weighting $H_{\text{sole}}$, prioritizing single-maintainer critical packages (achieving up to 88.04\% coverage).
  \end{itemize}
\end{itemize}

\subsection{2. Algorithmic Implementations and Exhaustive Benchmarking}
\begin{itemize}
  \item \textbf{Core Algorithm Pipeline (\nolinkurl{scripts/study_structural_interventions.py}):}
  \begin{itemize}
    \item Implementation of Greedy Set Cover heuristic for coverage.
    \item Implementation of Exact Union solver for transitive risk mass.
    \item Implementation of Direct Exclusion-Oriented heuristic for multi-maintainer exclusion.
    \item Evaluation across 6 standard policy budgets: $B \in \{10, 25, 50, 100, 200, 500\}$.
  \end{itemize}
  \item \textbf{Ablation and Sensitivity Benchmarking (Task 03 \& Task 23):}
  \begin{itemize}
    \item Executed 38 selection runs across 6 budgets (228 metric rows) isolating the impact of reciprocal scoring, candidate caps, and deterministic tie breaking.
    \item Evaluated 20 empirical candidate pools and 12 synthetic instances across budgets 1, 3, and 5 (96 settings, 288 method comparisons).
    \item Demonstrated that the direct exclusion rule matches 51/60 empirical and 35/36 synthetic restricted optima.
  \end{itemize}
  \item \textbf{Statistical Hypothesis Testing and Monte Carlo Simulations:}
  \begin{itemize}
    \item Conducted 6,000 random budget simulations across 200 distinct random seeds.
    \item Calculated Monte Carlo Standard Errors (MCSE) and non-parametric effect sizes (Cliff's $\delta = 1.0, p < 0.005$).
  \end{itemize}
\end{itemize}

\subsection{3. Data Mining, Dependency Graphs, and SQLite Infrastructure}
\begin{itemize}
  \item \textbf{Primary Ecosystem Graph Construction:}
  \begin{itemize}
    \item Mined transitive dependencies and metadata across 7,158 GitHub-linked npm endpoints and 12,217 maintainer accounts.
    \item Resolved strongly connected components (SCCs: 82/68 cycles) and dependency depth frontiers.
  \end{itemize}
  \item \textbf{Dataset Hygiene and Parser Disagreement Resolution (Task 02):}
  \begin{itemize}
    \item Audited 658 parser disagreements between npm registry manifests and repository \nolinkurl{package.json} configurations.
    \item Discovered and quantified the 42.4\% "version continuity drop" occurring between undated repository HEAD observations and immutable release versions.
  \end{itemize}
  \item \textbf{100\% Programmatic SQLite Extraction (Task 12):}
  \begin{itemize}
    \item Completely purged hardcoded LaTeX tables in the manuscript.
    \item Built automated SQLite extraction scripts guaranteeing exact database-to-manuscript data consistency (reconciling maintainer reach for \nolinkurl{qix} to exactly 1,476 dependents).
  \end{itemize}
  \item \textbf{Release-Authority Delta Protocol (Task 09):}
  \begin{itemize}
    \item Extracted 45 temporal release-authority snapshots with 97.8\% predecessor metadata availability.
    \item Reconstructed SemVer dependency graph with 1,848 edges and 99.89\% resolution accuracy.
  \end{itemize}
\end{itemize}

\subsection{4. Empirical and Real-World Security Validations}
\begin{itemize}
  \item \textbf{Retrospective Security Incident Validation (Task 04):}
  \begin{itemize}
    \item Systematically audited 49 documented npm security incidents; mapped 16 matching compromised packages into the study graph.
    \item Empirically proved that Exact Union captures the infamous account takeover of maintainer \textit{qix} at \textbf{Rank \#8} (\#5 in version graph), whereas Greedy (>100) and Portfolio (\#2,298) completely fail.
    \item Discovered and quantified the \textbf{85\%--92\% phantom blast-radius inflation} caused by unpinned transitive analysis on foundational packages like \nolinkurl{chalk} and \nolinkurl{ansi-styles}.
  \end{itemize}
  \item \textbf{Prospective Fresh Cohort Replication (Task 23):}
  \begin{itemize}
    \item Executed a clean, prospective data collection from 100 frozen roots: 94 dependency graphs, 1,641 exact nodes, 4,357 version-edge rows, and 1,051 accounts.
    \item Re-optimized on 65 common repository endpoints and 1,036 common accounts to confirm that algorithm behavioral trade-offs hold across independent observation epochs.
  \end{itemize}
  \item \textbf{The 120-Endpoint Human Audit Protocol (Task 23 \& Current):}
  \begin{itemize}
    \item Designed a formal stratified probability sampling frame: 120 repositories sampled across 4 source-cardinality strata (30 per stratum) from 7,158 endpoints.
    \item Captured 172 registry-request records and 115 commit-pinned snapshots.
    \item Engineered the offline \nolinkurl{Evidence_Index.html} dossier linking all source texts, manifests, workflows, and lockfiles.
  \end{itemize}
\end{itemize}
